            <button onclick="sendMessage()">Send</button>

    <script>
        // Frontend Logic to handle UI interactions
        const chatHistory = document.getElementById('chatHistory');
        const userInput = document.getElementById('userInput');

        // Allow sending message with Enter key
        userInput.addEventListener('keypress', function (e) {
            if (e.key === 'Enter') sendMessage();
        });

        function appendMessage(text, sender) {
            const msgDiv = document.createElement('div');
            msgDiv.className = `message ${sender}`;
            msgDiv.textContent = text;
            chatHistory.appendChild(msgDiv);
            
            // Auto-scroll to the newest message
            chatHistory.scrollTop = chatHistory.scrollHeight;
        }

        function sendMessage() {
            const text = userInput.value.trim();
            if (!text) return;

            // 1. Display User Message
            appendMessage(text, 'user');
            userInput.value = '';

            // 2. Simulate Agent Processing Delay
            setTimeout(() => {
                // In a real application, you would make an HTTP request here:
                // fetch('http://localhost:3000/api/chat', { method: 'POST', body: JSON.stringify({ message: text }) })
                
                let simulatedResponse = "";
                
                if (text.toLowerCase().includes("printer")) {
                    simulatedResponse = "I see you're dealing with a printer issue. The last time this happened, your successful fix was: 'Restart the router and reconnect via IP 192.168.1.15'.";
                } else if (text.toLowerCase().includes("expense")) {
                    simulatedResponse = "Expense successfully categorized and queued for the local ledger.";
                } else {
                    simulatedResponse = "Processing new task. Generating a novel solution...";
                }

                // 3. Display Agent Response
                appendMessage(simulatedResponse, 'agent');
            }, 800);
        }
    </script>
